cmake_minimum_required(VERSION 3.25)
project(TaiJiAura VERSION 1.0.0 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "TaiJiAura requires Windows and the MSVC toolchain.")
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build, for example cmake --preset windows-debug.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(desktopball_options INTERFACE)
target_compile_definitions(desktopball_options INTERFACE
    UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(desktopball_options INTERFACE /W4 /permissive- /utf-8 /EHsc)
add_subdirectory(src)
add_subdirectory(apps)
install(TARGETS DesktopBall RUNTIME DESTINATION .)

option(DESKTOPBALL_BUILD_UNINSTALL_LAUNCHER "Build the installer uninstall.exe entry point" OFF)
if(DESKTOPBALL_BUILD_UNINSTALL_LAUNCHER)
    add_subdirectory(packaging)
endif()
